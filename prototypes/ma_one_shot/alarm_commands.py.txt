    @api_command("player_queues/alarm_capabilities", required_scope=Scope.QUEUES_CONTROL)
    @handle_play_action
    async def alarm_capabilities(self, queue_id: str) -> dict[str, Any]:
        """Versioned scheduled-alarm contract; legacy test APIs are not sufficient."""
        result = await self.briefing_test_capabilities(queue_id, preserve_queue=True)
        queue_data = self._queue_data[queue_id]
        queue = queue_data.queue
        repeat_mode = getattr(queue.repeat_mode, "value", queue.repeat_mode)
        return {
            **result,
            "protocol": 2,
            "features": [
                "selected_track", "speech_delay", "music_only", "volume_ramp",
                "preserve_queue", "restore_playback", "queue_mode_lease",
            ],
            "queue_modes": {
                "repeat_mode": repeat_mode,
                "autoplay_override": queue_data.autoplay_override,
                "autoplay_enabled": queue.autoplay_enabled,
            },
        }

    def _valid_alarm_queue_modes(self, queue_modes: Any) -> bool:
        """Validate the read-only queue preferences HA persisted before dispatch."""
        if not isinstance(queue_modes, dict):
            return False
        repeat_modes = {
            getattr(mode, "value", mode) for mode in RepeatMode
        }
        return (
            queue_modes.get("repeat_mode") in repeat_modes
            and (queue_modes.get("autoplay_override") is None
                 or type(queue_modes.get("autoplay_override")) is bool)
            and type(queue_modes.get("autoplay_enabled")) is bool
        )

    async def _restore_alarm_queue_modes(self, queue_id: str, queue_modes: Any) -> bool:
        """Restore only settings still at the temporary alarm overrides."""
        if not self._valid_alarm_queue_modes(queue_modes):
            return False
        queue_data = self._queue_data.get(queue_id)
        if queue_data is None:
            return False
        queue = queue_data.queue
        restored = False
        off_value = getattr(RepeatMode.OFF, "value", RepeatMode.OFF)
        if (queue_modes["repeat_mode"] != off_value
                and getattr(queue.repeat_mode, "value", queue.repeat_mode) == off_value):
            repeat_mode = next(
                mode for mode in RepeatMode
                if getattr(mode, "value", mode) == queue_modes["repeat_mode"]
            )
            await self.set_repeat(queue_id, repeat_mode)
            restored = True
        if queue_modes["autoplay_enabled"] and not queue.autoplay_enabled:
            queue_data.autoplay_override = queue_modes["autoplay_override"]
            self._resolve_default_toggles(queue_data)
            self.signal_update(queue_id)
            restored = True
        return restored

    @api_command("player_queues/alarm_start", required_scope=Scope.QUEUES_CONTROL)
    @handle_play_action
    async def alarm_start(
        self, queue_id: str, run_id: str, track_uri: str, message: str = "",
        tts_engine: str = "", language: str | None = None, voice: str | None = None,
        music_gain: float = 0.25, speech_delay: int = 0, volume_level: int = 20,
        queue_modes: dict[str, Any] | None = None,
    ) -> dict[str, Any]:
        """Start a selected library song, optionally with one finite delayed briefing."""
        self._check_player_permission(queue_id)
        capability = await self.alarm_capabilities(queue_id)
        if not capability["ready"]:
            raise InvalidDataError("; ".join(capability["reasons"]))
        if queue_modes is None:
            queue_modes = capability["queue_modes"]
        if (not self._valid_alarm_queue_modes(queue_modes)
                or queue_modes != capability["queue_modes"]):
            raise InvalidDataError("The Music Assistant queue settings changed; check the alarm before trying again")
        if (not isinstance(track_uri, str) or not track_uri.startswith("library://track/")
                or not track_uri.removeprefix("library://track/").isascii()
                or not track_uri.removeprefix("library://track/").isdigit()):
            raise InvalidDataError("Choose a Music Assistant library track")
        if type(volume_level) is not int or not 1 <= volume_level <= 100:
            raise InvalidDataError("Alarm volume must be between 1 and 100 percent")
        levels = MixLevels(music_gain, 0.7, speech_delay)
        levels.validate()
        if not 0 < music_gain <= 0.3:
            raise InvalidDataError("Music gain must be above zero and at most 30 percent")
        if not isinstance(message, str) or len(message) > 4096:
            raise InvalidDataError("Invalid alarm message")
        if message and tts_engine not in {engine["uid"] for engine in capability["engines"]}:
            raise InvalidDataError("Select this Home Assistant voice through its Music Assistant provider")
        mixer = self.mass.streams.audio.briefings
        if mixer.has_run(queue_id, run_id):
            raise InvalidDataError("This alarm start was already attempted")
        # Reserve before preparation/enqueue, including failures and lost responses.
        mixer.reserve_alarm_start(queue_id, run_id)
        prepared = None
        dispatch_started = False
        if message:
            prepared = await prepare_briefing(
                self.mass, run_id, message, tts_engine, language, voice, levels=levels)
        try:
            media = await self.mass.music.get_item_by_uri(track_uri)
            if not isinstance(media, Track) or not media.available:
                raise InvalidDataError("The selected alarm track is unavailable")
            # Use MA's normal QueueItem builder, but not normal ADD's tail shuffle
            # or queue-source replacement policy. Append to the current live list.
            item = build_queue_item(queue_id, media)
            player = self.mass.players.get_player(queue_id)
            if player is None:
                raise PlayerUnavailableError("Alarm player became unavailable")
            queue = self._queue_data[queue_id].queue
            prior = queue.current_item
            resume = (queue.state == PlaybackState.PLAYING
                      and player.state.playback_state == PlaybackState.PLAYING
                      and player.state.active_source in (None, queue_id)
                      and prior is not None
                      and any(entry.queue_item_id == prior.queue_item_id
                              for entry in self._queue_data[queue_id].items))
            owner = {"run_id": run_id, "item_id": item.queue_item_id, "preserve_queue": True,
                     "music_gain": music_gain, "volume_level": volume_level,
                     "original_volume": player.state.volume_level,
                     "queue_modes": dict(queue_modes),
                     "resume_item_id": prior.queue_item_id if resume and prior is not None else None,
                     "resume_position": max(0, int(queue.corrected_elapsed_time)) if resume else 0}
            key = (queue_id, run_id)
            mixer.alarm_segments[key] = owner
            # Keep the user's effective queue preferences normally; suspend them only
            # while this guarded alarm segment owns playback.
            if queue.repeat_mode != RepeatMode.OFF:
                await self.set_repeat(queue_id, RepeatMode.OFF)
            if queue.autoplay_enabled:
                self.set_autoplay(queue_id, False)
            dispatch_started = True
            self.update_items(queue_id, [*self._queue_data[queue_id].items, item])
            # Music-only starts also consume their id before dispatch. Failed or
            # uncertain starts must never be blindly submitted a second time.
            if prepared is None:
                mixer.claim_music_run(queue_id, run_id)
            await self._play_index(queue_id, item.queue_item_id, 0, False,
                                   briefing=prepared, briefing_owner=owner)
            return {"protocol": 2, "status": "playing"}
        except BaseException:
            mixer.cancel(queue_id, run_id)
            if not dispatch_started:
                await self._restore_alarm_queue_modes(queue_id, queue_modes)
                mixer.alarm_segments.pop((queue_id, run_id), None)
            raise
        finally:
            if prepared is not None:
                prepared.source = ""

    @api_command("player_queues/alarm_control", required_scope=Scope.QUEUES_CONTROL)
    @handle_play_action
    async def alarm_control(
        self, queue_id: str, run_id: str, action: str, volume_level: int | None = None,
        queue_modes: dict[str, Any] | None = None,
    ) -> dict[str, Any]:
        """Release one alarm segment, restoring only the music it interrupted."""
        self._check_player_permission(queue_id)
        if action not in ("status", "snooze", "restart", "stop", "restore_modes"):
            raise InvalidDataError("Unknown alarm action")
        if volume_level is not None:
            if action != "restart" or type(volume_level) is not int or not 1 <= volume_level <= 100:
                raise InvalidDataError("Only a stopped alarm restart can reset volume (1-100)")
        mixer = self.mass.streams.audio.briefings
        data = self._queue_data[queue_id]
        player = self.mass.players.get_player(queue_id)
        key = (queue_id, run_id)
        if action == "restore_modes":
            owner = mixer.owner(queue_id, run_id, data.session_id)
            if owner is not None:
                return {"protocol": 2, "owned": True, "restored_modes": False}
            segment = mixer.alarm_segments.get(key)
            saved_modes = segment.get("queue_modes") if segment else queue_modes
            restored = await self._restore_alarm_queue_modes(queue_id, saved_modes)
            return {"protocol": 2, "owned": False, "restored_modes": restored}
        if action in ("snooze", "stop") and key in mixer.alarm_releases:
            # A lost response must not stop the restored music or restore twice.
            return dict(mixer.alarm_releases[key])
        if action in ("snooze", "stop"):
            mixer.reject_run(queue_id, run_id)
        owner = mixer.owner(queue_id, run_id, data.session_id)
        current = data.queue.current_item
        if (owner is None or player is None
                or (not owner.get("preserve_queue") and len(data.items) != 1)
                or not any(item.queue_item_id == owner["item_id"] for item in data.items)
                or (data.queue.state != PlaybackState.IDLE
                    and (current is None or current.queue_item_id != owner["item_id"]))
                or player.state.active_source not in (None, queue_id)
                or player.state.active_group or player.state.synced_to
                or player.state.type == PlayerType.GROUP
                or len(player.state.group_members or ()) > 1):
            # Ownership loss never authorizes stop/restore. A stale alarm item
            # can still be removed if it is neither current nor buffered.
            segment = mixer.alarm_segments.get(key)
            if action in ("snooze", "stop") and segment is not None:
                buffered = self.get_item(queue_id, data.queue.index_in_buffer)
                upcoming = data.queue.next_item
                protected = {item.queue_item_id for item in (current, buffered, upcoming) if item is not None}
                if segment["item_id"] not in protected:
                    items = [item for item in data.items if item.queue_item_id != segment["item_id"]]
                    if current is not None:
                        data.queue.current_index = next(
                            (i for i, item in enumerate(items) if item.queue_item_id == current.queue_item_id), None)
                    if buffered is not None:
                        data.queue.index_in_buffer = next(
                            (i for i, item in enumerate(items) if item.queue_item_id == buffered.queue_item_id), None)
                    self.update_items(queue_id, items)
                await self._restore_alarm_queue_modes(
                    queue_id, segment.get("queue_modes", queue_modes)
                )
                mixer.alarm_segments.pop(key, None)
                result = {"protocol": 2, "owned": False, "released": True}
                mixer.alarm_releases[key] = result
                return dict(result)
            if action in ("snooze", "stop") and queue_modes is not None:
                restored = await self._restore_alarm_queue_modes(queue_id, queue_modes)
                return {
                    "protocol": 2, "owned": False, "released": True,
                    "restored_modes": restored,
                }
            return {"protocol": 2, "owned": False}
        if action == "restart":
            # Natural song loops only. Snooze wakes use a fresh alarm_start and
            # capture whatever is playing then, with a new one-shot speech ID.
            if data.queue.state != PlaybackState.IDLE:
                raise InvalidDataError("Wait for the alarm song to end before looping it")
            if volume_level is not None:
                owner.update(volume_level=volume_level, ramp_suspended=False)
            elif owner.get("ramp_checked"):
                actual = player.state.volume_level
                if type(actual) is not int or not 0 <= actual <= 100:
                    raise InvalidDataError("Check the speaker volume before restarting its song")
                if actual != owner["volume_level"]:
                    owner.update(volume_level=actual, ramp_suspended=True, original_volume=None)
            await self._play_index(queue_id, owner["item_id"], 0, False, briefing_owner=owner)
        elif action in ("snooze", "stop"):
            return await self._alarm_release(queue_id, run_id, owner)
        return {"protocol": 2, "owned": True, "state": data.queue.state.value}

    async def _alarm_release(
        self, queue_id: str, run_id: str, owner: dict[str, Any],
    ) -> dict[str, Any]:
        """Release verified ownership while the caller holds the queue lock."""
        mixer = self.mass.streams.audio.briefings
        data = self._queue_data[queue_id]
        player = self.mass.players.get_player(queue_id)
        assert player is not None
        session_id = data.session_id
        mixer.cancel(queue_id, run_id)
        await self._handle_stop(queue_id)
        if data.session_id is not None or mixer.owner(queue_id, run_id, session_id) is None:
            return {"protocol": 2, "owned": False}
        mixer.cancel_queue(queue_id)
        # Filter the current list, not the original queue. Same-song copies
        # and all additions, deletions and reordering by the user survive.
        data.queue.current_index = None
        data.queue.index_in_buffer = None
        data.queue.current_item = None
        self.update_items(queue_id, [
            item for item in data.items if item.queue_item_id != owner["item_id"]
        ])
        key = (queue_id, run_id)
        result = {"protocol": 2, "owned": True, "state": "idle", "restored": False}
        mixer.alarm_releases[key] = result
        mixer.alarm_segments.pop(key, None)
        try:
            if (player.state.volume_level == owner["volume_level"]
                    and type(owner["original_volume"]) is int):
                await self.mass.players.cmd_volume_set(queue_id, owner["original_volume"])
            prior_id = owner.get("resume_item_id")
            if (data.session_id is None
                    and player.state.active_source in (None, queue_id)
                    and not player.state.active_group and not player.state.synced_to
                    and len(player.state.group_members or ()) <= 1
                    and prior_id and any(item.queue_item_id == prior_id for item in data.items)):
                await self._play_index(queue_id, prior_id, owner["resume_position"], False)
                result.update(state="playing", restored=True)
            await self._restore_alarm_queue_modes(queue_id, owner.get("queue_modes"))
        except BaseException:
            # Record uncertainty before returning/raising. Never retry a restore
            # dispatch after it may have reached the player.
            result["restoration_failed"] = True
            raise
        return dict(result)

    @api_command("player_queues/alarm_volume", required_scope=Scope.QUEUES_CONTROL)
    @handle_play_action
    async def alarm_volume(
        self, queue_id: str, run_id: str, volume_level: int | None = None,
    ) -> dict[str, Any]:
        """Check ownership and manual changes under the queue lock, without replaying media."""
        self._check_player_permission(queue_id)
        if volume_level is not None and (type(volume_level) is not int or not 1 <= volume_level <= 100):
            raise InvalidDataError("Alarm volume must be between 1 and 100 percent")
        status = await self.alarm_control(queue_id, run_id, "status")
        if not status["owned"]:
            return {**status, "protocol": 2}
        mixer = self.mass.streams.audio.briefings
        data = self._queue_data[queue_id]
        owner = mixer.owner(queue_id, run_id, data.session_id)
        player = self.mass.players.get_player(queue_id)
        assert owner is not None
        assert player is not None
        owner["ramp_checked"] = True
        mixer.bind_owner(queue_id, data.session_id, owner)
        actual = player.state.volume_level
        if type(actual) is not int or not 0 <= actual <= 100 or actual != owner["volume_level"]:
            # A manual change owns the level now, including at the next song loop.
            # Never restore the old pre-alarm level over that manual choice.
            owner.update(ramp_suspended=True, original_volume=None)
            if type(actual) is int and 0 <= actual <= 100:
                owner["volume_level"] = actual
            mixer.bind_owner(queue_id, data.session_id, owner)
        suspended = bool(owner.get("ramp_suspended"))
        if volume_level is not None and not suspended and status["state"] == PlaybackState.PLAYING.value:
            owner["volume_level"] = volume_level
            # Record intent before dispatch: an uncertain response is never retried.
            mixer.bind_owner(queue_id, data.session_id, owner)
            await self.mass.players.cmd_volume_set(queue_id, volume_level)
            actual = volume_level
        return {**status, "protocol": 2, "volume_level": actual, "suspended": suspended}
